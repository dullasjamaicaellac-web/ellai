My Java Project

CC102 - Final Performance Task (PeTa)

This project is a Number Guessing Game developed using Java. The player must guess the secret number between 1 and 20 within five attempts. It uses basic Java concepts such as Scanner, Random, if-else statements, and while loops.

Java Source Code

import java.util.Scanner;
import java.util.Random;

public class NumberGuessingGame {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        Random random = new Random();

        int secretNumber = random.nextInt(20) + 1;
        int guess;
        int attempts = 0;
        int maxAttempts = 5;

        System.out.println("=================================");
        System.out.println("       NUMBER GUESSING GAME");
        System.out.println("=================================");
        System.out.println("Guess the secret number from 1 to 20.");
        System.out.println("You have " + maxAttempts + " attempts.");

        while (attempts  maxAttempts) {
            System.out.print("\nEnter your guess: ");
            guess = input.nextInt();
            attempts++;

            if (guess  1 || guess > 20) {
                System.out.println("Please enter a number from 1 to 20.");
                attempts--;
            } else if (guess == secretNumber) {
                System.out.println("Congratulations! You guessed the number!");
                System.out.println("Number of attempts: " + attempts);
                break;
            } else if (guess  secretNumber) {
                System.out.println("Too low! Try a higher number.");
            } else {
                System.out.println("Too high! Try a lower number.");
            }

            if (attempts == maxAttempts) {
                System.out.println("\nGame over! You used all your attempts.");
                System.out.println("The secret number was " + secretNumber + ".");
            }
        }

        System.out.println("\nThank you for playing!");
        input.close();
    }
}
